ros2 topic pub /nav/phase std_msgs/String "{data: object}" -1
ros2 topic pub /nav/goal geometry_msgs/PoseStamped "header:
  frame_id: map
pose:
  position: {x: 1.0, y: 0.5, z: 0.0}
  orientation: {w: 1.0}" -1

  frame_id: map
pose:
  position: {x: -0.5, y: 0.2, z: 0.0}
  orientation: {w: 1.0}" -1
